def show_query(query, params):
    sql = query
 
    for p in params:
 
        if isinstance(p, datetime):
            value = f"'{p.strftime('%Y-%m-%d %H:%M:%S')}'"
        else:
            value = f"'{p}'"
 
        sql = sql.replace("?", value, 1)
 
    print("\nFINAL SQL:")
    print(sql)
    print("-" * 100)
